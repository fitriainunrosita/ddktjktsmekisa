Kode Google Apps Script (Code.gs)

Guru dapat menghubungkan aplikasi ini ke Google Spreadsheet agar jawaban dan nilai tersimpan otomatis serta siswa dapat dihapus sinkron:

  1. Buka Google Sheets baru & Beri nama: Rekap Nilai TJKT X - Ibu Fitri.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus kode bawaan dan tempel kode di bawah ini, lalu klik Deploy > New deployment.
  4. Pilih jenis Web app, setting Who has access ke Anyone.
  5. Salin Web App URL dan tempelkan pada kolom URL di panel guru.
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Fitur Hapus Siswa Otomatis dari Panel Guru
    if (data.action === "DELETE_STUDENT") {
      var rows = sheet.getDataRange().getValues();
      for (var i = rows.length - 1; i >= 1; i--) {
        if (rows[i][1] === data.studentName && rows[i][2] === data.studentClass) {
          sheet.deleteRow(i + 1);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({result: "deleted"})).setMimeType(ContentService.MimeType.JSON);
    }
    
    // Auto Create Headers jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Nama Lengkap", "Kelas", "Bab 1", "Bab 2", "Bab 3", "Bab 4", "Final Exam", "Status Akhir"]);
      sheet.getRange(1, 1, 1, 9).setFontWeight("bold").setBackground("#0891b2").setFontColor("#ffffff");
    }
    
    // Cari apakah siswa sudah ada di baris spreadsheet
    var rows = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var r = 1; r < rows.length; r++) {
      if (rows[r][1] === data.studentName && rows[r][2] === data.studentClass) {
        rowIndex = r + 1;
        break;
      }
    }
    
    var time = new Date().toLocaleString("id-ID", { timeZone: "Asia/Jakarta" });
    
    if (rowIndex > 0) {
      // Update data baris siswa
      sheet.getRange(rowIndex, 1).setValue(time);
      if (data.bab1 !== undefined) sheet.getRange(rowIndex, 4).setValue(data.bab1);
      if (data.bab2 !== undefined) sheet.getRange(rowIndex, 5).setValue(data.bab2);
      if (data.bab3 !== undefined) sheet.getRange(rowIndex, 6).setValue(data.bab3);
      if (data.bab4 !== undefined) sheet.getRange(rowIndex, 7).setValue(data.bab4);
      if (data.finalExam !== undefined) sheet.getRange(rowIndex, 8).setValue(data.finalExam);
      if (data.status !== undefined) sheet.getRange(rowIndex, 9).setValue(data.status);
    } else {
      // Baris baru
      sheet.appendRow([
        time,
        data.studentName,
        data.studentClass,
        data.bab1 || 0,
        data.bab2 || 0,
        data.bab3 || 0,
        data.bab4 || 0,
        data.finalExam || 0,
        data.status || "Belum Tuntas"
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
